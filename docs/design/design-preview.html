项目目录结构

src/ 目录

│   │   ├── day01.json
│   │   ├── ...
│   │   └── day30.json
│   ├── npcs.json            # NPC 基础数据
│   ├── factions.json        # 门派数据
│   ├── martial-arts.json    # 武学数据

│   ├── screens/             # 全屏场景

数据流架构

核心数据流（自上而下）
JSON Data Layer
events/*.json   npcs.json   martial-arts.json   endings.json
↓ 加载 & 校验
Engine Layer（纯逻辑）
TimeEngine → EventEngine → FavorEngine → EndingEngine   |   CombatEngine   |   FlagEngine
↓ 状态更新（immutable）
React State（Zustand）
GameState { day, timeslot, favors, flags, health, qi, martialArts... }
↓ useGameState / useEventRunner
UI Layer（React）
GameScreen（双栏）  |   CombatScreen   |   RelationGraph   |   EndingScreen

事件脚本数据结构

JSON 驱动 | 每天一个文件 | 引擎解析触发条件并执行效果

事件脚本格式（day01.json 示例）

// day01.json — 第 1 天全部事件 { "day": 1, "events": [ { "id": "day01_arrival", "type": "story", "trigger": { "timeslot": "dawn", "location": "zhongyuan", "conditions": [] }, "scene": { "background": "zhongyuan_mountain", "description": "清晨的嵩山古道，薄雾未散……", "ambient": "morning_birds" }, "dialogues": [ { "speaker": "narrator", "text": "你背着行囊，踏上了嵩山古道。" }, { "speaker": "old_wei", "text": "年轻人，第一次来嵩山？", "emotion": "smile" } ], "choices": [ { "id": "friendly", "text": "「是，前辈。请指路。」", "effects": [ { "type": "favor", "npc": "old_wei", "trust": +5, "intimacy": +3 }, { "type": "flag", "key": "wei_friendly_intro" } ] }, { "id": "aloof", "text": "「不用了，我自己能找到。」", "effects": [ { "type": "favor", "npc": "old_wei", "awe": +5, "trust": -3 } ] }, { "id": "curious", "text": "「前辈看起来不像普通人。」", "conditions": [ { "type": "flag", "key": "wei_friendly_intro" } ], "effects": [ { "type": "favor", "npc": "old_wei", "trust": +8, "awe": +3 }, { "type": "unlock", "what": "old_wei_identity_hint" } ] } ] } ] }

GameState 类型定义

interface GameState { currentDay: number; // 1-30 currentTimeslot: Timeslot; // "dawn" | "noon" | "dusk" | "night" currentLocation: Location; // 五大地理区域之一 health: number; // 生命值 maxHealth: number; // 生命上限 qi: number; // 气力 maxQi: number; // 气力上限 attack: number; defense: number; speed: number; npcFavors: Map<NPC_ID, FavorDims>; // 四维好感度 factionReputation: Map<Faction_ID, number>; // 0-100 learnedMartialArts: MartialArt[]; equippedSkills: MartialArt[4]; flags: Set<string>; // 关键抉择标记 endingsUnlocked: Ending[]; isSecondPlaythrough: boolean; // 二周目 } interface FavorDims { trust: number; // 信任 0-100 intimacy: number; // 亲密 0-100 awe: number; // 敬畏 0-100 fear: number; // 忌惮 0-100 } type Timeslot = "dawn" | "noon" | "dusk" | "night"; type Location = "zhongyuan" | "jiangnan" | "saibei" | "shuzhong" | "xiyu";
